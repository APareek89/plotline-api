%% 02 index build and sync flow

flowchart TD
  A["Read JSONL<br/>[FUNCTION]<br/>in: one JSON object per line<br/>out: Pydantic KBChunk list"]:::fn
  B{"Corpus ceiling<br/>[FUNCTION]<br/>chunk count at most 20,000?"}:::dec
  C["Reject build<br/>[DATA · error]<br/>in: invalid or oversized corpus<br/>out: actionable failure"]:::term
  D["Embed documents<br/>[LIBRARY · FastEmbed ONNX]<br/>in: chunk text<br/>out: normalized float32 matrix"]:::data
  E["Tokenize corpus<br/>[FUNCTION]<br/>in: chunk text<br/>out: lowercase term lists"]:::fn
  F["Build BM25<br/>[LIBRARY · rank-bm25]<br/>in: token lists<br/>out: BM25Okapi index"]:::data
  G["Write NPZ and pickle<br/>[FUNCTION]<br/>in: matrix, IDs, BM25<br/>out: cached artifacts"]:::fn
  H["Hash manifest<br/>[FUNCTION]<br/>in: JSONL and artifacts<br/>out: model, dim, count, SHA-256"]:::fn
  I{"Sync mode<br/>[FUNCTION]<br/>dry run?"}:::dec
  J["Validation report<br/>[DATA · CLI result]<br/>in: manifest<br/>out: no AWS mutation"]:::term
  Q{"AWS cost fence<br/>[FUNCTION]<br/>all plotline tables provisioned and total at most 10 RCU and 10 WCU?"}:::dec
  R["Refuse sync<br/>[DATA · error]<br/>in: unsafe billing or capacity<br/>out: no S3 or DDB mutation"]:::term
  K{"Remote bundle current<br/>[FUNCTION]<br/>manifest and all object hashes match?"}:::dec
  L["Upload bundle<br/>[LIBRARY · boto3 S3]<br/>in: four local files<br/>out: private S3 objects"]:::data
  M{"Chunk hash current<br/>[FUNCTION]<br/>DynamoDB item hash matches?"}:::dec
  N["Batch write one item<br/>[LIBRARY · boto3 DynamoDB]<br/>in: serialized chunk item<br/>out: stored source ID or error after 8 unprocessed retries"]:::data
  O["WCU throttle<br/>[FUNCTION]<br/>in: item size and elapsed time<br/>out: sleep at least units divided by 1 WCU"]:::fn
  P["Sync report<br/>[DATA · CLI result]<br/>in: uploaded, written, skipped<br/>out: idempotency evidence"]:::term

  Q -->|"yes"| K

%% 03 search and source resolution flow
%% files: src/plotline_rag/schema.py, src/plotline_rag/search/hybrid.py, src/plotline_rag/service.py
%% granularity: full
flowchart TD
  A["Validate request<br/>[LIBRARY · Pydantic]<br/>in: query at most 4,000 chars, k at most 50, filters, corpus<br/>out: normalized SearchRequest"]:::data
  B["Filter candidates<br/>[FUNCTION]<br/>in: chunks and metadata filters<br/>out: eligible row indexes"]:::fn
  C{"Candidates exist<br/>[FUNCTION]<br/>eligible count greater than zero?"}:::dec
  D["Empty results<br/>[DATA · API result]<br/>in: no candidates<br/>out: empty list plus manifest"]:::term
  E["Embed query<br/>[LIBRARY · FastEmbed ONNX]<br/>in: query text<br/>out: normalized 384-d vector"]:::data
  F["Dense cosine<br/>[LIBRARY · NumPy]<br/>in: matrix and query vector<br/>out: cosine score per candidate"]:::data
  G["BM25 scores<br/>[LIBRARY · rank-bm25]<br/>in: query terms and BM25 index<br/>out: lexical score per candidate"]:::data
  H["Reciprocal-rank fusion<br/>[FUNCTION]<br/>in: dense and BM25 ranks<br/>out: fused score using rank constant 60"]:::fn
  I{"Rerank requested<br/>[FUNCTION]<br/>rerank equals true?"}:::dec
  J["V1 reserved error<br/>[DATA · HTTP 400]<br/>in: unsupported rerank request<br/>out: explicit error"]:::term
  K["Evidence results<br/>[DATA · API result]<br/>in: top k fused rows<br/>out: IDs, text, scores, manifest"]:::term
  L["Validate source IDs<br/>[FUNCTION]<br/>in: up to 100 unique IDs<br/>out: existence lookups"]:::fn
  M{"Online resolver<br/>[FUNCTION]<br/>S3-backed online startup?"}:::dec
  N["DynamoDB consistent gets<br/>[LIBRARY · boto3]<br/>in: at most 100 source_id_full keys<br/>out: existing IDs or logged HTTP 503"]:::data
  O["Local cache set<br/>[DATA · in-memory IDs]<br/>in: offline source IDs<br/>out: existing IDs"]:::data
  P["Resolution response<br/>[DATA · API result]<br/>in: existing ID set<br/>out: resolved and unresolved lists"]:::term
